# -*- coding: utf-8 -*- # ff_hack_overlay.py - Web điều khiển + Overlay ESP (cửa sổ nổi) # Đóng gói thành .exe bằng PyInstaller: pyinstaller --onefile --windowed ff_hack_overlay.py import ctypes import struct import math import threading import time import sys import tkinter as tk from tkinter import font from flask import Flask, render_template_string, jsonify, request # ==================== HACK ENGINE ==================== try: from pymem import Pymem import pydirectinput except: print("Cài: pip install pymem pydirectinput flask") sys.exit(1) PROCESS_NAME = b"aow_exe.exe" # đổi nếu dùng MEmu pm = None BASE = 0 # OFFSET (CẬP NHẬT THEO GAME) OFF = { "LOCAL_PLAYER": 0x14A2B8C0, "ENTITY_LIST": 0x14A3F000, "POS_X": 0x2C0, "POS_Y": 0x2C4, "POS_Z": 0x2C8, "VIEW_ANGLE": 0x1A40, "HEALTH": 0x1A8, "TEAM_ID": 0x1D0, "IS_VISIBLE": 0x1E8, "HEAD_BONE": 0x520, "NO_RECOIL": 0x23A0 } class Vector3(ctypes.Structure): _fields_ = [("x", ctypes.c_float), ("y", ctypes.c_float), ("z", ctypes.c_float)] class Vector2(ctypes.Structure): _fields_ = [("x", ctypes.c_float), ("y", ctypes.c_float)] def init_hack(): global pm, BASE try: pm = Pymem(PROCESS_NAME) BASE = pm.process_base.lpBaseOfDll return True except: return False def read_float(a): return pm.read_float(a) def read_int(a): return pm.read_int(a) def read_bool(a): return pm.read_bool(a) def read_vec3(a): return Vector3(*struct.unpack("fff", pm.read_bytes(a, 12))) def write_float(a,v): pm.write_float(a, ctypes.c_float(v)) def write_vec2(a,v): pm.write_bytes(a, struct.pack("ff", v.x, v.y)) def get_local(): return pm.read_int(BASE + OFF["LOCAL_PLAYER"]) def get_entity(i): p = pm.read_int(BASE + OFF["ENTITY_LIST"] + i*4) return p if p else None # Trạng thái aimbot_on = False norecoil_on = False esp_on = False # Dữ liệu ESP (để overlay vẽ) esp_data = [] # list of (x, y, dist, health) def scan_esp(): global esp_data if not esp_on: return local = get_local() if not local: return local_pos = read_vec3(local + OFF["POS_X"]) local_team = read_int(local + OFF["TEAM_ID"]) data = [] for i in range(1, 101): ent = get_entity(i) if not ent: continue if read_int(ent + OFF["TEAM_ID"]) == local_team: continue if read_int(ent + OFF["HEALTH"]) <= 0: continue if not read_bool(ent + OFF["IS_VISIBLE"]): continue pos = read_vec3(ent + OFF["POS_X"]) dist = math.sqrt((pos.x-local_pos.x)**2 + (pos.y-local_pos.y)**2 + (pos.z-local_pos.z)**2) # Chuyển sang tọa độ màn hình (giả định) screen_x = (pos.x * 0.8 + pos.z * 0.2 + 1) * 960 screen_y = (pos.y * -0.7 + pos.z * 0.3 + 1) * 540 data.append((screen_x, screen_y, dist, read_int(ent + OFF["HEALTH"]))) esp_data = data def run_aimbot(): if not aimbot_on: return local = get_local() if not local: return local_pos = read_vec3(local + OFF["POS_X"]) local_team = read_int(local + OFF["TEAM_ID"]) cur_angle = read_vec3(local + OFF["VIEW_ANGLE"]) best_fov = 40.0 best_angle = None for i in range(1, 101): ent = get_entity(i) if not ent: continue if read_int(ent + OFF["TEAM_ID"]) == local_team: continue if read_int(ent + OFF["HEALTH"]) <= 0: continue if not read_bool(ent + OFF["IS_VISIBLE"]): continue head = read_vec3(ent + OFF["HEAD_BONE"]) dx = head.x - local_pos.x dy = head.y - local_pos.y dz = head.z - local_pos.z yaw = math.atan2(dy, dx) * 57.2958 dist = math.sqrt(dx*dx + dy*dy + dz*dz) pitch = -math.atan2(dz, dist) * 57.2958 fov = abs(yaw - cur_angle.x) + abs(pitch - cur_angle.y) if fov < best_fov: best_fov = fov best_angle = Vector2(yaw, pitch) if best_angle: smooth = 6.0 new_x = cur_angle.x + (best_angle.x - cur_angle.x) / smooth new_y = cur_angle.y + (best_angle.y - cur_angle.y) / smooth write_vec2(local + OFF["VIEW_ANGLE"], Vector2(new_x, new_y)) pydirectinput.click(button='left') def run_norecoil(): if norecoil_on: local = get_local() if local: write_float(local + OFF["NO_RECOIL"], 0.0) # ==================== OVERLAY (cửa sổ nổi) ==================== class Overlay: def __init__(self): self.root = tk.Tk() self.root.title("FF ESP Overlay") self.root.geometry("1920x1080") self.root.attributes('-alpha', 0.4) # trong suốt self.root.attributes('-topmost', True) self.root.attributes('-fullscreen', True) self.root.overrideredirect(True) self.root.config(bg='black') self.canvas = tk.Canvas(self.root, bg='black', highlightthickness=0) self.canvas.pack(fill=tk.BOTH, expand=True) self.font = font.Font(family="Arial", size=12, weight="bold") def update_overlay(self): self.canvas.delete("all") if esp_on: for (x, y, dist, hp) in esp_data: if 0 < x < 1920 and 0 < y < 1080: self.canvas.create_rectangle(x-20, y-30, x+20, y+10, outline='lime', width=2) self.canvas.create_text(x, y-35, text=f"{int(dist)}m", fill='cyan', font=self.font) self.canvas.create_text(x+25, y-10, text=f"HP:{hp}", fill='red', font=self.font) self.root.after(50, self.update_overlay) # 20 FPS def run(self): self.update_overlay() self.root.mainloop() # ==================== WEB SERVER ==================== app = Flask(__name__) HTML = ''' FF Hack

⚡ FF CONTROL

Bật/tắt – có cửa sổ nổi ESP
🎯 Aimbot
🔫 No Recoil
👁️ ESP (overlay)
Chưa bật
Overlay luôn hiển thị khi ESP bật
''' @app.route('/') def index(): return render_template_string(HTML) @app.route('/toggle/', methods=['POST']) def toggle(f): global aimbot_on, norecoil_on, esp_on if f=='aimbot': aimbot_on = not aimbot_on elif f=='norecoil': norecoil_on = not norecoil_on elif f=='esp': esp_on = not esp_on else: return jsonify({'error':1}),400 return jsonify({'success':True, 'status':{'aimbot':aimbot_on,'norecoil':norecoil_on,'esp':esp_on}}) @app.route('/status') def status(): return jsonify({'aimbot':aimbot_on,'norecoil':norecoil_on,'esp':esp_on}) # ==================== MAIN ==================== def hack_loop(): while True: if pm and BASE: run_aimbot() run_norecoil() scan_esp() time.sleep(0.015) if __name__ == '__main__': print("[+] Kết nối game...") if not init_hack(): print("[-] Không tìm thấy game! Vẫn chạy nhưng hack ko hoạt động.") else: print("[+] Đã kết nối.") # Chạy thread hack threading.Thread(target=hack_loop, daemon=True).start() # Chạy overlay trong thread riêng overlay = Overlay() threading.Thread(target=overlay.run, daemon=True).start() # Chạy web server (chặn main) print("[+] Web tại http://localhost:5000") app.run(host='0.0.0.0', port=5000, debug=False)
